add_executable(NexoraEditor main.cpp)
if(MSVC)
  target_compile_definitions(NexoraEditor PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
target_link_libraries(NexoraEditor PRIVATE Nexora::Editor Nexora::Runtime Nexora::Core)
if(NEXORA_ENABLE_EDITOR_GRAPHICAL_SHELL)
  if(WIN32)
    target_link_libraries(NexoraEditor PRIVATE ole32 shell32 uuid)
  endif()
  target_link_libraries(NexoraEditor PRIVATE Nexora::EditorImGui Nexora::Presentation)
  if(NEXORA_ENABLE_2D)
    target_link_libraries(NexoraEditor PRIVATE Nexora::TwoD)
  endif()
  target_compile_definitions(NexoraEditor PRIVATE NEXORA_EDITOR_GRAPHICAL_SHELL=1)
  if(BUILD_TESTING)
    target_compile_definitions(NexoraEditor PRIVATE NEXORA_EDITOR_EXTENSION_UI_TEST_ACCESS=1)
    target_include_directories(NexoraEditor PRIVATE "${PROJECT_SOURCE_DIR}/Engine/EditorImGui/src")
  endif()
endif()
nexora_apply_defaults(NexoraEditor)
