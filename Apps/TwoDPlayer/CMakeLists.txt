set(NEXORA_2D_GAME_PROJECT "game.nx2game" CACHE STRING "Executable-relative game descriptor filename")
function(nexora_add_2d_game target descriptor)
  if(NOT target MATCHES "^[A-Za-z][A-Za-z0-9_]*$")
    message(FATAL_ERROR "2D game target must be an ASCII CMake identifier")
  endif()
  if(NOT descriptor MATCHES "^[A-Za-z0-9_-]+[.]nx2game$")
    message(FATAL_ERROR "2D game descriptor must be a contained ASCII .nx2game filename")
  endif()
  add_executable(${target} WIN32 "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/GameMain.cpp")
  target_link_libraries(${target} PRIVATE NexoraTwoDPlayerHost)
  target_compile_definitions(${target} PRIVATE "NEXORA_2D_GAME_PROJECT=\"${descriptor}\"")
  if(WIN32)
    target_link_libraries(${target} PRIVATE shell32 user32)
  endif()
  if(MINGW)
    target_link_options(${target} PRIVATE -municode)
  endif()
  nexora_apply_defaults(${target})
endfunction()
if(NEXORA_2D_GAME_TARGET)
  nexora_add_2d_game("${NEXORA_2D_GAME_TARGET}" "${NEXORA_2D_GAME_PROJECT}")
endif()
