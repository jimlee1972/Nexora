include(NexoraTarget)
set(NEXORA_CRYPTOGRAPHY_BACKEND "AUTO" CACHE STRING "Signature provider: AUTO, OPENSSL or NONE")
set_property(CACHE NEXORA_CRYPTOGRAPHY_BACKEND PROPERTY STRINGS AUTO OPENSSL NONE)
if(NOT NEXORA_CRYPTOGRAPHY_BACKEND MATCHES "^(AUTO|OPENSSL|NONE)$")
  message(FATAL_ERROR "NEXORA_CRYPTOGRAPHY_BACKEND must be AUTO, OPENSSL or NONE")
endif()
set(NEXORA_CRYPTOGRAPHY_OPENSSL_AVAILABLE OFF CACHE INTERNAL "Selected cryptographic backend" FORCE)
if(NEXORA_CRYPTOGRAPHY_BACKEND STREQUAL "OPENSSL")
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
elseif(NEXORA_CRYPTOGRAPHY_BACKEND STREQUAL "AUTO" AND NOT CMAKE_CROSSCOMPILING)
  find_package(OpenSSL 3.0 QUIET COMPONENTS Crypto)
endif()
add_library(NexoraCryptography ${NEXORA_MODULE_LIBRARY_TYPE} src/Signature.cpp)
add_library(Nexora::Cryptography ALIAS NexoraCryptography)
target_include_directories(NexoraCryptography PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_link_libraries(NexoraCryptography PUBLIC Nexora::Foundation)
target_compile_definitions(NexoraCryptography PRIVATE NEXORA_CRYPTOGRAPHY_EXPORTS)
if(NEXORA_LINK_MODE STREQUAL "Monolithic")
  target_compile_definitions(NexoraCryptography PUBLIC NEXORA_CRYPTOGRAPHY_STATIC=1)
endif()
if(TARGET OpenSSL::Crypto AND (NEXORA_CRYPTOGRAPHY_BACKEND STREQUAL "OPENSSL" OR
    (NEXORA_CRYPTOGRAPHY_BACKEND STREQUAL "AUTO" AND NOT CMAKE_CROSSCOMPILING)))
  target_link_libraries(NexoraCryptography PRIVATE OpenSSL::Crypto)
  target_compile_definitions(NexoraCryptography PRIVATE NEXORA_CRYPTOGRAPHY_OPENSSL=1)
  set(NEXORA_CRYPTOGRAPHY_OPENSSL_AVAILABLE ON CACHE INTERNAL "Selected cryptographic backend" FORCE)
  message(STATUS "Nexora Cryptography: OpenSSL ${OPENSSL_VERSION}")
else()
  message(STATUS "Nexora Cryptography: unavailable provider (verification fails closed)")
endif()
nexora_apply_defaults(NexoraCryptography)
