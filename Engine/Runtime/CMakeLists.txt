include(NexoraTarget)
add_library(NexoraRuntime ${NEXORA_MODULE_LIBRARY_TYPE}
  src/GameplayModuleHost.cpp
  src/Runtime.cpp)
if(NEXORA_ENABLE_ASSET_PIPELINE)
  target_sources(NexoraRuntime PRIVATE src/AssetPipeline.cpp)
endif()
if(NEXORA_ENABLE_EDITOR_SDK)
  target_sources(NexoraRuntime PRIVATE src/EditorSdk.cpp)
endif()
add_library(Nexora::Runtime ALIAS NexoraRuntime)
target_include_directories(NexoraRuntime PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_link_libraries(NexoraRuntime PUBLIC Nexora::Renderer)
if(NEXORA_ENABLE_EDITOR_SDK AND NOT WIN32)
  target_link_libraries(NexoraRuntime PRIVATE ${CMAKE_DL_LIBS})
endif()
target_compile_definitions(NexoraRuntime PRIVATE NEXORA_RUNTIME_EXPORTS)
if(NEXORA_ENABLE_ASSET_PIPELINE)
  target_compile_definitions(NexoraRuntime PUBLIC NEXORA_ASSET_PIPELINE_ENABLED=1)
else()
  target_compile_definitions(NexoraRuntime PUBLIC NEXORA_ASSET_PIPELINE_ENABLED=0)
endif()
if(NEXORA_ENABLE_EDITOR_SDK)
  target_compile_definitions(NexoraRuntime PUBLIC NEXORA_EDITOR_SDK_ENABLED=1)
else()
  target_compile_definitions(NexoraRuntime PUBLIC NEXORA_EDITOR_SDK_ENABLED=0)
endif()
if(NEXORA_ENABLE_SCENE_RENDERING)
  target_compile_definitions(NexoraRuntime PUBLIC NEXORA_SCENE_RENDERING_ENABLED=1)
else()
  target_compile_definitions(NexoraRuntime PUBLIC NEXORA_SCENE_RENDERING_ENABLED=0)
endif()
if(NEXORA_LINK_MODE STREQUAL "Monolithic")
  target_compile_definitions(NexoraRuntime PUBLIC NEXORA_RUNTIME_STATIC=1)
endif()
nexora_apply_defaults(NexoraRuntime)
